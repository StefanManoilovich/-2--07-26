using namespace std;

int main() {
    int a = 4, b = 5;
    
    cout << "До: a = " << a << ", b = " << b << endl;
    
    // меняем значения местами без третьей переменной
    a = a + b;
    b = a - b;
    a = a - b;
    
    cout << "После: a = " << a << ", b = " << b << endl;
    
    return 0;
}
Задание №4
#include <iostream>

using namespace std;

int main() {
    int A[] = {1, 3, 5, 7, 9};
    int B[] = {2, 4, 6, 8, 10, 12};
    
    int n = sizeof(A) / sizeof(A[0]);
    int m = sizeof(B) / sizeof(B[0]);
    
    int i = 0, j = 0;
    
    // слияние двух упорядоченных массивов в один поток вывода
    for (int k = 0; k < n + m; k++) {
        if (i >= n) {
            cout << B[j] << " ";
            j++;
        } else if (j >= m) {
            cout << A[i] << " ";
            i++;
        } else if (A[i] < B[j]) {
            cout << A[i] << " ";
            i++;
        } else {
            cout << B[j] << " ";
            j++;
        }
    }
    cout << endl;
    
    return 0;
}

#include <iostream>

using namespace std;

int main() {
    int arr[] = {0, 1, 2, 3, 5, 6, 7};
    int n = 7;
    
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += arr[i];
    }
    
    int full_sum = n * (n + 1) / 2;
    int missing = full_sum - sum;
    
    cout << "Недостающее число: " << missing << endl;
    
    return 0;
}
